import math
import cadquery as cq


def _f(p, k, default=0.0):
    v = p.get(k)
    if v is None:
        return float(default)
    return float(v)


def _cham(p, label):
    leg = _f(p, label, 0.0)
    if leg <= 0.0:
        return 0.0, 0.0
    ang = p.get(label + '__angleDeg')
    if ang is None:
        ang = 45.0
    return leg, leg * math.tan(math.radians(float(ang)))


def _dims(p):
    A = _f(p, 'A')
    B = _f(p, 'B')
    C = _f(p, 'C')
    D = _f(p, 'D')
    E = _f(p, 'E')
    K = _f(p, 'K')

def _profile(g):

    RB = g['RB']
    RK = g['RK']
    RE = g['RE']
    RL = g['RL']
    RN = g['RN']
    C = g['C']
    D = g['D']
    M = g['M']
    Jx = g['Jx']
    Jr = g['Jr']
    Fx = g['Fx']
    Fr = g['Fr']
    Gx = g['Gx']
    Gr = g['Gr']
    pts = []

    def add(x, z):
        if pts and abs(pts[-1][0] - x) < 1e-9 and abs(pts[-1][1] - z) < 1e-9:
            return
        pts.append((x, z))

    zc = -(C + M)
    zend = zc - 5.0
    add(RA - Jr, 0.0)
    add(RA, -Jx)
    add(RA, -C)

        add(RE, -C)
        add(RE, zc)

    add(RN, zend)
    add(RK + Gr, zend)

        add(RK, -D)
        add(RB, -D)
        add(RB, -Fx)
    else:
        add(RK, -Fx)

def build_part(p):

    pts = _profile(g)

    body = cq.Workplane(plane).polyline(pts).close().revolve(360.0, (0, 0), (0, 1))
    return body
